\begin{frame}{Overview}
  \begin{enumerate}
    \item Established borrowing priors use one study-level parameter. Compatibility with an external control can vary across patient profiles.
    \item LRC-BART adds a sparse discrepancy ensemble to the pooled BART surface, with a spike-and-slab commensurate prior on each discrepancy leaf.
    \item We calibrate a variance-based prior ESS for the standardized control mean. Its ceiling depends on the RWD fit at trial profiles.
    \item Simulations with regional shifts, and a 30-patient single-arm myeloma trial with a registry control.
  \end{enumerate}
\end{frame}

\section{Motivation}
\begin{frame}{External controls and transportability}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Potential gain}
        \begin{itemize}
          \item More control observations
          \item Greater precision
          \item Smaller concurrent control arms
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{alertblock}{Why pooling can mislead}
        \begin{itemize}
          \item Different eligibility and care pathways
          \item Different covariate distributions
          \item Measured and unmeasured confounding
        \end{itemize}
      \end{alertblock}
  \end{columns}
  \vspace{0.5em}
  \centering
  \textbf{External controls may agree with the trial in one patient group and differ in another. A single borrowing parameter cannot distinguish these groups.}
\end{frame}

\begin{frame}{Established priors use a study-level borrowing parameter}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Meta-analytic predictive prior \citep{Neuenschwander2010MAP,schmidli2014robust}}
        Study parameters are exchangeable, $\theta_j\mid\mu,\tau^2\sim\Normal(\mu,\tau^2)$; the robust version adds a vague component that suppresses borrowing under conflict.
      \end{block}
      \begin{block}{Commensurate prior \citep{Hobbs2011,Hobbs2012}}
        $\theta\mid\theta_0\sim\Normal(\theta_0,1/\tau)$ with a spike-and-slab hyperprior on the commensurability $\tau$.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Power prior \citep{IbrahimChen2000}}
        Historical likelihood raised to $a_0$.
      \end{block}
      \begin{alertblock}{Common feature}
        One heterogeneity or discount scale applies to the whole population, without depending on the patient's covariates. With a single historical study the three forms nearly coincide \citep{RoverFriede2025}.
      \end{alertblock}
  \end{columns}
  \vspace{0.4em}
  \centering
  \textbf{We allow borrowing to vary across covariate regions learned from the data.}
\end{frame}

\section{Model}
\begin{frame}{The trial-control mean adds a discrepancy to the external mean}
  \small
  RCT controls ($D_i=1$) and RWD controls ($D_i=2$), covariates $\bx_i$, outcome $y_i$:
  \[
    \boxed{\;y_i=f(\bx_i)+\mathbb{I}\{D_i=1\}\,g(\bx_i)+\varepsilon_i,\qquad \varepsilon_i\sim\Normal(0,\sigma^2_{D_i})\;}
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{$f$: external-control response mean}
        Standard BART, $H_f=50$ trees, fitted to both sources. The RWD follow $f$.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{$g$: discrepancy ensemble}
        Small and shallow, $H_g=5$ trees, split prior $(0.5,3)$, informed by the RCT controls only. The RCT controls follow $f_1=f+g$.
      \end{block}
  \end{columns}
  \vspace{0.4em}
  \begin{itemize}\footnotesize
    \item The decomposition follows Bayesian causal forests \citep{Hahn2020BCF}, with source membership replacing treatment.
    \item The discrepancy is $g=f_1-f$. Shrinking it toward zero centers the RCT-control mean $f_1$ at the external-control mean $f$.
  \end{itemize}
\end{frame}

\begin{frame}{Each discrepancy leaf has a spike-and-slab prior}
  \footnotesize
  The zero-centered mixture describes each leaf contribution to $g=f_1-f$:\par\vspace{-0.5em}
  $\theta_{h\ell}$ is the contribution of leaf $(h,\ell)$ to $g$:
  \[
    \theta_{h\ell}\given w,\tau_0^2,\tau_1^2
    \stackrel{\mathrm{iid}}{\sim} w\,\Normal(0,\tau_0^2)+(1-w)\,\Normal(0,\tau_1^2).
  \]
  Equivalently, $z_{h\ell}\given w\stackrel{\mathrm{iid}}{\sim}\Ber(w)$ selects a zero-mean normal with variance $\tau_{1-z_{h\ell}}^2$. We use $w\sim\mathrm{Beta}(1,1)$, $\tau_0^2\sim\text{scaled-Inv-}\chi^2(\nu_0,s_0^2)$ truncated to $(0,\tau_1^2)$, and $\tau_1^2$ fixed by the range rule.
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Spike, $z=1$: the commensurate component}
        Shrinks a discrepancy contribution toward zero, drawing the RCT response mean toward $f(\bx)$. The ESS calibration sets the leaf scale $\tau_0$.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Slab, $z=0$: the vague component}
        A free local offset on the outcome scale, corresponding to a robust-MAP vague component within a leaf.
      \end{block}
  \end{columns}
  \vspace{0.3em}
  \begin{block}{Special cases}
    $w=1$: commensurate BART (C-BART), one learned scale. \quad $w=1,\ \tau_0^2\to0$: BART on the pooled controls (BART-CP). \quad $w=0$: a free source offset, the tree analogue of BART with the source indicator (BART-PP).
  \end{block}
\end{frame}

\begin{frame}{The induced prior is centered at $f(\bx)$}
  \small
  \begin{itemize}
    \item At a fixed profile $\bx$, each discrepancy tree contributes one leaf:
      $g(\bx)=\sum_{h=1}^{H_g}\theta_{h,\ell_h(\bx)}$, where $\ell_h(\bx)$ is the reached leaf.
    \item Integrating the leaf indicators gives a mixture over the number $m$ of reached spike leaves:
  \end{itemize}
  \[
  f_1(\bx)\given f,T_g,w,\tau_0^2,\tau_1^2
  \sim\sum_{m=0}^{H_g}\binom{H_g}{m}w^m(1-w)^{H_g-m}
  \Normal\!\left(f(\bx),m\tau_0^2+(H_g-m)\tau_1^2\right).
  \]
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{All spike: $m=H_g$}
        Variance $H_g\tau_0^2$: the commensurate component shrinks $f_1(\bx)$ toward $f(\bx)$.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{All slab: $m=0$}
        Variance $H_g\tau_1^2$: a larger departure is permitted, with the same center $f(\bx)$.
      \end{block}
  \end{columns}
  \vspace{0.3em}
  \begin{itemize}\footnotesize
    \item $T_g$ denotes the discrepancy partitions; every component is centered at the external mean $f(\bx)$.
    \item This is a prior for the latent control mean. Observation variance is separate.
  \end{itemize}
\end{frame}

\begin{frame}{Separate ensembles allow different shrinkage priors}
  \small
  \begin{itemize}
    \item Separate ensembles let us choose different partition and shrinkage priors for the external mean $f$ and the discrepancy $g$.
    \item We use $H_f=50$ trees for the outcome surface and $H_g=5$ shallow trees for the discrepancy.
    \item At fixed leaf variances and partitions, fewer discrepancy trees reduce the total all-spike variance at a profile. The slab range rule also depends on $H_g$.
  \end{itemize}
  \begin{block}{What the prior-density comparison establishes}
    Appendix G.6 minimizes over integer numbers of slab leaves and retains contributions from the remaining spike leaves. It compares joint prior densities at constrained modes. Posterior indicator probabilities also depend on the likelihood and partitions.
  \end{block}
  \footnotesize The simulations assess this specification. They do not establish that shared-leaf models cannot detect regional discrepancies.
\end{frame}

\begin{frame}{Tree moves use exact leaf marginals}
  \footnotesize
  Conditionally on $(\tau_0^2,\tau_1^2,w)$ the leaf prior is a finite Gaussian mixture, which gives closed-form factors for each tree move:
  \[
    M_f=(1+\lambda_f^2A)^{-1/2}\exp\Bigl\{\frac{\lambda_f^2B^2}{2(1+\lambda_f^2A)}\Bigr\},
    \qquad
    M_g=\frac{w\,\phi(\bar y_1;0,\tau_0^2+v_1)+(1-w)\,\phi(\bar y_1;0,\tau_1^2+v_1)}{\phi(\bar y_1;0,v_1)},
  \]
  with $A,B$ the precision-weighted counts and sums of both sources in an $f$ leaf, and $\bar y_1$, $v_1=\sigma_1^2/n_1$ the RCT residual mean and its variance in a $g$ leaf.
  \begin{itemize}
    \item Grow, prune and change moves use the exact collapsed ratios, so the sampler targets the exact posterior without a surrogate likelihood.
    \item Leaf draws are conjugate; $z$ is drawn first, then $\theta$; $\tau_0^2$ and $w$ update from the leaf means and states.
    \item For right-censored outcomes, log-normal AFT data augmentation leaves the other updates unchanged.
    \item One sweep costs a standard BART on $n_1+n_2$ observations plus a five-tree ensemble on $n_1$.
  \end{itemize}
\end{frame}

\section{Borrowing calibration}
\begin{frame}{Prior effective sample size at the estimand}
  \footnotesize
  For $\mu=N^{-1}\sum_i f_1(\bx_i)$, $V_\mu^f$ is the RWD-only posterior variance. The working calibration is:
  \[
    \mathrm{ESS}_0(s_0^2)=\hat\sigma_1^2\,\E_{\tau_0^2}\Bigl[\frac{1}{V_\mu^f+c_gH_g\tau_0^2}\Bigr].
  \]
  \begin{itemize}
    \item $c_g$ averages squared leaf proportions before inversion; the scale law here is untruncated.
    \item A normal prior gives the ELIR variance ratio \citep{neuenschwander2020}; this BART/scale-mixture summary is not exact marginal ELIR.
    \item Ceiling: $\hat\sigma_1^2/V_\mu^f$, including BART regularization. Equality to scaled $n_2$ requires one fixed tree, matched proportions and a flat prior.
    \item The Pr-rule requires the target in 95\% of blocks. A target at or above the whole-chain ceiling instead triggers a smallest-scale override.
  \end{itemize}
  \begin{block}{Realized ESS diagnostic}
    With target leaf proportions $a_{h\ell}$, use $V_g=\sum_{h,\ell}a_{h\ell}^2\{z_{h\ell}\tau_0^2+(1-z_{h\ell})\tau_1^2\}$, then average $\sigma_1^2/(V_\mu^f+V_g)$. Standardization can span several leaves per tree.
  \end{block}
\end{frame}

\begin{frame}{Conditional theoretical guarantees}
  \footnotesize
  Leaf results condition on $f$, the other discrepancy trees, partitions, scales and $0<w<1$. Write $v=\sigma_1^2/n$ for a leaf's RCT residual-mean variance.
  \begin{block}{Theorem 1: bounded local influence}
    Relative to the slab estimate, $\sup_{\bar y}|I(\bar y)|\le\sqrt{2v\max\{\log C,\tfrac12\}}$, and the shift vanishes at a Gaussian rate under growing conflict. The C-BART and BART-CP leaf shifts are linear and unbounded.
  \end{block}
  \begin{block}{Theorem 2: local detection and map consistency}
    The leaf spike probability decreases with $|\bar y|$. With fixed partitions, $f$ fixed, a representable discrepancy and growing local counts, $\mathcal B_c(x)\to\mathbb I\{|d|<c\}$ for $|d|\ne c$. A fixed positive spike scale need not identify compatibility through its indicator.
  \end{block}
  \begin{block}{Theorem 3: the working calibration}
    Under the untruncated law, $\mathrm{ESS}_0$ is continuous, convex and strictly decreasing from its ceiling to zero. Truncation gives a positive lower limit. Grid-selection consistency requires the stated moment and separation conditions; ceiling overrides are separate.
  \end{block}
\end{frame}

\begin{frame}{The borrowing map}
  \footnotesize
  At each patient profile $x$, we report
  \[
    \mathcal B_c(x)=P\bigl(|g(x)|<c\given\text{data}\bigr),
  \]
  the posterior probability that the local discrepancy is below a threshold $c$ fixed in advance on the outcome scale, together with the posterior mean and interval of $g(x)$ and the pointwise ESS.
  \begin{itemize}
    \item $c=0.5$ in the simulations, one third of the RCT residual SD for Gaussian outcomes.
    \item The all-spike probability $B(x)$ is kept as a diagnostic: its prior scale is $w^{H_g}$ and at a fixed positive spike scale, it need not identify compatibility as RCT counts grow.
    \item In single-arm mode the data do not inform $g$, so the compatibility map is flat. The local ESS map describes variation across profiles.
  \end{itemize}
  \begin{alertblock}{Conditions for the map result}
    The map results require fixed $g$ partitions, a representable discrepancy, growing local counts and $f$ fixed; the result does not cover the posterior over partitions. The simulations show separation of shifted and compatible regions under the full posterior.
  \end{alertblock}
\end{frame}

\begin{frame}[t]{Single-arm trials cannot learn a control discrepancy}
  \footnotesize
  Trial contributes treated subjects only ($n_T=200$ Gaussian; $30$ or $200$ survival); the $300$ RWD controls are the sole control source; LRC-BART in single-arm mode with $w=1$.
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}{No confounding}
        BART-CP and LRC-BART have Gaussian RMSE $0.16$; the survival RMST ratio at $n_T=30$ has RMSE $0.16$ to $0.19$ with coverage $0.92$ to $0.97$.
      \end{block}
    \column{0.49\textwidth}
      \begin{alertblock}{Measured confounding}
        Every method is biased: Gaussian ATE bias $1.68$ (LM-CP), $0.41$ (BART-CP), $0.42$ (LRC-BART); there are no internal controls to identify a discrepancy, and RWD overlap is limited.
      \end{alertblock}
  \end{columns}
  \vspace{0.3em}
  \begin{itemize}
    \item The single-arm ceiling is $27$ to $34$ under Sc2, so an absolute target of $100$ is capped and LRC-BART approximates pooling with $w=1$; the fractional target still widens the interval (coverage $0.82$ to $0.99$).
    \item With $w<1$, slab variance $(1-w)H_g\tau_1^2$ adds uncertainty: SD $1.4$ for the ATE and $10$ to $34$ for the median ratio. Zero-mean $g$ preserves linear log-time means, but survival-ratio means also change. Primary analysis: $w=1$.
    \item Hierarchical regressions: ATE SD $8$ under the stated hyperprior.
  \end{itemize}
\end{frame}

\section{Takeaways}
\begin{frame}{Main findings}
  \begin{enumerate}
    \item The RCT control mean is centered on the external mean through a spike-and-slab discrepancy ensemble.
    \item Closed-form leaf marginals support posterior-invariant tree updates; this does not establish finite-chain mixing.
    \item Variance-based ESS calibration concerns the standardized control mean. The calculation specifies its ceiling and approximation convention.
    \item Conditional leaf influence and map results require the stated fixed-surface and partition assumptions.
    \item Gaussian Sc1 RMSE is about $16\%$ lower than BART-PP's. Regional conflict produces informative compatibility summaries, but also surface losses and excessive borrowing at moderate shifts.
  \end{enumerate}
\end{frame}

\begin{frame}{Limitations and next steps}
  \begin{itemize}
    \item Full-posterior map consistency over learned partitions is unproved; the Sc4 surface loss persists away from the boundary.
    \item Calibration uses a working precision summary. Exact marginal ELIR would require a different calculation and new calibration analyses.
    \item Multiple external sources require joint calibration. Additive ceilings have not been established.
    \item A single-arm study cannot learn the control discrepancy. The application remains a descriptive comparison with post-induction ASCT adjustment.
    \item Confirmatory use requires operating-characteristic evaluation, including Type I error and power, and a pre-specified role for the map.
  \end{itemize}
  \begin{block}{How to interpret the borrowing summaries}
    The compatibility map, prior pointwise ESS and realized regional ESS answer different questions. These summaries do not give an observed count of external patients used at each profile.
  \end{block}
\end{frame}
